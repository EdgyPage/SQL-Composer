def busy_jobs() -> Statement:
    """Only the jobs with 3 runs or more (HAVING).

    WHERE picks rows before they are counted, so it can't test a count. HAVING tests each
    group after GROUP_BY has counted it.

    return statement(
        SELECT(job_runs.job_id, AS(count_rows(), "runs")),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
        HAVING(at_least(count_rows(), 3)),

jobs_that_never_ran()

def jobs_that_never_ran() -> Statement:
    """Every job with no run on these days (in SQL, an anti-join).

    LEFT_JOIN keeps every job, and a job with no run gets NULL in job_runs' columns, so
    is_null(job_runs.run_id) keeps just those. The days' filter goes in ON=, not in WHERE: in
    WHERE it would throw away the very jobs with no runs, whose dt is NULL too.
    """
    return statement(
        SELECT(jobs.job_id, jobs.job_name),
        FROM(jobs),
        LEFT_JOIN(
            job_runs,
            ON=all_of(equals(job_runs.job_id, jobs.job_id),
                      between(job_runs.dt, FIRST_DAY, LAST_DAY)),
            # A job has many runs. That is fine here: the jobs kept are those with none.
            many_matches=True,
        ),
        WHERE(is_null(job_runs.run_id)),
    )

jobs_that_ran()

def jobs_that_ran() -> Statement:
    """Every job with at least one run on these days, each job once (in SQL, a semi-join).

    A JOIN straight to job_runs would repeat a job once per run. So first make the list of
    job_ids that ran, each once, with SELECT_DISTINCT, then JOIN to that list.
    """
    ran = derived("ran", statement(
        SELECT_DISTINCT(job_runs.job_id),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
    ))
    return statement(
        SELECT(jobs.job_id, jobs.job_name),
        FROM(jobs),
        JOIN(ran, ON=equals(ran.job_id, jobs.job_id)),
    )

labelled_runs()

def labelled_runs() -> Statement:
    """Each run with a readable status and a length label (in SQL, COALESCE and CASE WHEN).

    A run still going has no status yet (NULL), which fill_null shows as RUNNING. if_else
    gives one value where its condition holds and the other where it doesn't.
    """
    return statement(
        SELECT(
            job_runs.run_id,
            AS(fill_null(job_runs.status, "RUNNING"), "status"),
            AS(if_else(more_than(job_runs.duration_mins, 20), "long", "short"), "length"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
    )

FROM ops.job_runs AS job_runs

def outcomes_per_job() -> Statement:
    """One row per job, with a column for each outcome (a pivot).

    count_rows(where=...) counts only the rows where its condition holds, so each column
    counts one kind of run. A run still going is counted with is_null, since equals can't
    find NULL. A TEST run is in none of the three columns, so they needn't add up to the runs.
    """
    return statement(
        SELECT(
            job_runs.job_id,
            AS(count_rows(where=equals(job_runs.status, "SUCCESS")), "succeeded"),
            AS(count_rows(where=equals(job_runs.status, "FAILED")), "failed"),
            AS(count_rows(where=is_null(job_runs.status)), "still_running"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
    )

failed_share_per_job()

def failed_share_per_job() -> Statement:
    """Each job's failed runs as a percent of its runs: one count divided by another.

    Divide after GROUP_BY, as one count by another, not an average of per-row values. Times
    100.0 for a percent; / keeps the decimals even between two counts, in Hive
    (DIV is the one that drops them). A division by 0 gives NULL, which pandas shows as NaN;
    COUNT(*) of a group is never 0, so it can't happen here.
    """
    return statement(
        SELECT(
            job_runs.job_id,
            AS(count_rows(where=equals(job_runs.status, "FAILED")) * 100.0 / count_rows(),
               "failed_percent"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
    )

WHERE
  job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'

Build a Saved table: create it, write a day, add to a day, backfill, rebuild.

Why: a result that is slow to work out, or read by many notebooks, is cheaper to work out once a day and save than to work out again every time someone reads it.

runs_to_review is partitioned by dt, and each write below fills one day's Partition, named with partition={"dt": day}. Each function below is one step. The Example database can't be written to, so the steps show their Hive but no result. """The Table reference of mart.runs_to_review, a Saved table the Worked examples write.

A Saved table's Table reference is filled in by hand, since the table doesn't exist until
create_table makes it from this: in a Project, project.new_table writes the file, and you fill
it in. It is partitioned by dt, and each day's Partition holds the runs someone should look at
that day. Its files live in a folder of its own, location=, a folder in the warehouse, not one
in your Project, written exactly as the Hive should name it. purge=True makes dropping it
delete its files too.
"""

from sqlglot_composer import Table

runs_to_review = Table(
    "mart.runs_to_review",
    columns={
        "run_id": "bigint",
        "job_id": "bigint",
        "dt": "string",  # the day the run happened, as yyyy-MM-dd
    },
    partitioned_by=["dt"],
    key=["run_id"],
    location="/data/your_team/mart.db/runs_to_review",
    purge=True,
)

def failed_runs(day: str = LAST_DAY) -> Statement:
    """Step 2: fill one day with that day's failed runs, replacing whatever the day held.

    table filled by one Statement alone can instead be written in one go: INSERT_OVERWRITE
    without partition=, its SELECT listing dt last, fills each day's Partition from the rows.

def careless(many_matches: bool = False) -> Statement:
    """What most people write first: it runs, and the Toolbox warns at the JOIN line.

    run_alerts has one row per alert, so each run is repeated once per alert it raised.
    careless(many_matches=True) pastes the Warning's opt-out, which silences it but still
    gives 150 minutes instead of 100.
    """
    return statement(
        SELECT(AS(sum_of(job_runs.duration_mins), "minutes"), AS(count_rows(), "alerts")),
        FROM(job_runs),
        JOIN(run_alerts, ON=equals(run_alerts.run_id, job_runs.run_id),
             many_matches=many_matches),
        WHERE(between(job_runs.dt, DAY, DAY), between(run_alerts.dt, DAY, DAY)),
    )

Table

A Table reference: a table's columns, its partition columns and its key.

Its columns become attributes, so job_runs.status is checked when your script runs: a typo stops with the list of real columns. Write one per table, generated once by write_table_reference(...) and then yours to edit.

  • columns maps each column to its Hive type as DESCRIBE, the command that lists a table's columns, prints it ("bigint", "decimal(10,2)"), or None when you don't know it. A typed column gets its values checked.
  • partitioned_by lists the columns the table is partitioned by, in the order Hive partitions it, such as ["dt"] or ["region", "load_id"], each a column in columns too; leave it out for a table that isn't partitioned. Each value of them is a folder of the table's files, a Partition, and a Statement reading the table must filter at least one of them (help(FROM)). A write fills the Partitions its rows hold, or the ones it names (help(INSERT_OVERWRITE)).
  • key lists the columns that pick out one row, the table's Key, for a table that has one; leave it out for one that hasn't. JOIN warns when a join doesn't use all of them; a write into a Saved table with a Key is refused unless the Toolbox can see each row it writes has a Key of its own (help(INSERT_OVERWRITE) says how); and create_table writes it into the table (help(create_table)). Hive never checks it.
  • does_not_add_up lists columns that are averages, ratios or distinct counts, which sum_of(...) and average_of(...) refuse to add up.
  • descriptions says in words what a column holds, such as {"status": "SUCCESS or FAILED"}, for the columns you want to describe. write_table_reference fills it from the comments DESCRIBE prints; it never changes the Hive.

A Saved table, one your Statements write into, also says how it is stored, which create_table writes and check_table_reference compares with the table. Of these, a Saved table needs only location; a table someone else owns has none of them:

  • location is the table's own folder, exactly as create_table writes it, such as "/data/your_team/mart.db/daily_runs": the Toolbox adds nothing to it. Every Saved table is an external table: its files are yours, in that folder, and the warehouse only keeps a record of where. Two tables in one folder would read each other's files, so a Saved table whose folder is another's, or inside another's, is refused.
  • purge=True makes dropping the table delete its files too. purge=False keeps them in its folder, where a table created again reads them back: at once, or, for a partitioned table, its Partitions once repair_partitions(t) lists them; left out, the warehouse's own choice, which keeps them, is neither written nor checked.
  • properties holds any other table properties, the notes on a table the warehouse keeps by name, such as {"made_by": "daily_pipeline"}.
  • stored_as is the file format, "ORC" (the usual one) or "PARQUET", and compression how its files are packed: "SNAPPY", or ORC's "ZLIB" or Parquet's "GZIP", among others.
  • bucketed_by with buckets spreads each Partition's rows into that many files by the values of those columns, and sorted_by sorts the rows in each, so a join on them can pair file with file. Leave them out unless the table is big and joined on those columns: changing them means writing the table again.

A column whose name is a Python word, such as from, is reached with getattr(t, "from"). You never type backticks: the Hive puts a name in them where it needs them, such as from, any or two words, and leaves a plain name, such as dt, as it is.

FULL_JOIN

Pair two tables' matching rows, and keep each table's rows with no match.

Use it to compare two records of the same things, such as the runs job_runs holds and the runs the scheduler started: each run comes out once, matched or not, with NULL in the columns of the side that lacks it, and fill_null(...) takes a value from whichever side has one. INSERT_OVERWRITE

Write rows into a Saved table, replacing the Partitions they fill, or all of it.

It goes first, before SELECT. Columns are matched to the Saved table by name, and the Toolbox puts them in the table's order, its partition columns last, as Hive fills them. Into a table with partitioned_by=, each row goes to the Partition its own values of the partition columns name, and INSERT_OVERWRITE replaces only the Partitions the rows fill: one they don't fill keeps what it held. Sent again, it replaces them again rather than adding to them, so a write can safely be sent twice; to add rows instead, use INSERT_INTO. If your warehouse runs Hive 2.3 or 3.1 under Tez (ask whoever looks after it), a write that now comes back empty may leave the old rows in place (HIVE-18702). INSERT_INTO

Add rows to a Saved table or its Partitions; sent twice, it adds twice.

It follows every rule of INSERT_OVERWRITE: it goes first, its columns are matched by name, it fills Partitions from the rows or names one with partition={...}, and it writes into the whole table for one with no partitioned_by=. The difference is what happens to the rows already there: INSERT_OVERWRITE replaces the Partitions it fills, INSERT_INTO keeps their rows and adds more. Use it when a Partition's rows come from more than one table: send INSERT_OVERWRITE first, from the first table, then INSERT_INTO from each other table. Sending the same INSERT_INTO twice adds its rows twice, so to redo it, start again from its INSERT_OVERWRITE. A Saved table's Key is checked as INSERT_OVERWRITE checks it, on the rows this write adds: rows the table holds already may share a Key with them, which check_key(t, query=..., where=...) finds once they are written. UNION_ALL

Stack the rows of Statements one after another, keeping every row.

Each Statement must return the same columns. Hive lines columns up by their place in SELECT, so it would quietly put one column under another of a different name; the Toolbox lines them up by name, and writes each Statement's columns in the first one's order. rollup

GROUP_BY(rollup(jobs.team, jobs.job_name)) gives the usual row for each team and job, as GROUP_BY(jobs.team, jobs.job_name) does, and as well, a subtotal row for each team, which totals over every job_name, so job_name is NULL in it, and the grand total, one row over all the rows, with team NULL too. window

The rows a calculation looks across for each row, passed to it as over=.

  • PARTITION_BY= makes the groups: each row looks only at the rows sharing its values. The word is SQL's, and has nothing to do with the Partitions a table is stored in. Left out, every row is in one group.
  • ORDER_BY= puts each group's rows in order, with descending(...) for largest first.
  • rows=(start, end) keeps only the rows near each one, counted from the row itself: negative before it, 0 the row itself, positive after it. None means no limit: every row before it as the start, every row after it as the end. rows=(-6, 0) is the row and the 6 before it; rows=(None, 0) every row up to it. It needs ORDER_BY. Left out, each row looks across its whole group.

PARTITION_BY and ORDER_BY take columns or a list of them, never a name given in SELECT. Each row keeps its own row, so a Window changes no row count, and needs no GROUP_BY. Hive can't test a Window's result in WHERE or HAVING of the Statement that works it out: work it out in a derived(...) table, and test it in the Statement that reads it. A Window over the totals of a GROUP_BY, such as each job's share of the day, goes the same way: total in a derived(...) table, and the Window in the Statement that reads it.

hive_settings

The SET lines of Session settings, by name, to send before your Statements.

Hive keeps a Session setting for the rest of the session, so send them first, then your Statements: run(hive_settings(...), execute=...) sends each SET line through your execute, as a call of its own, and gives back None. They hold only while your query and execute share one session across their calls: if each call opens a new session, each SET is gone before your Statement runs. show_hive prints them, and export_lineage, passed them with the Statements, writes them at the foot of both its files.

Name one or more sets. The Toolbox ships four:

  • "dynamic_partitions": hive.exec.dynamic.partition=true and hive.exec.dynamic.partition.mode=nonstrict. For a table partitioned by more than its day, such as by region too: a write gives the day, and Hive fills the other Partitions from each row's own values only with these on.
  • "map_joins": hive.auto.convert.join=true, hive.auto.convert.join.noconditionaltask=true and hive.ignore.mapjoin.hint=false. Hive then joins a small table by copying it to every worker, which skips sorting the big one, and chooses this by itself from the tables' statistics; and it heeds the hint that JOIN(..., broadcast=True) writes, which it ignores otherwise. explain(s) shows whether it made one: a Map Join in the plan.
  • "statistics_on_write": hive.stats.autogather=true and hive.stats.column.autogather=true. Each write then gathers the statistics of the day it writes, which Hive picks its join plans from, so they are never out of date. compute_statistics(t, day) counts a day written without them.
  • "merge_small_files": hive.merge.mapfiles=true, hive.merge.mapredfiles=true and hive.merge.tezfiles=true. A write then merges its small files into fewer, larger ones, since many small files slow down every later read.

Keep your own sets in a settings.py of your own, beside your notebook, each a name and its settings, such as HIVE_SETTINGS = {"nightly": {"hive.exec.parallel": True}}, and pass them as sets=. True and False are written true and false.

compute_statistics

The ANALYZE TABLE Statement, which counts a Saved table's rows again.

Hive picks how to join a table from its statistics: how many rows and bytes it holds, in each of its Partitions. Without them it guesses, and may sort two big tables to join them even when copying a small one to every worker would do; the Session setting "map_joins" lets Hive choose to copy it. run(compute_statistics(t), execute=...) reads the table's files, counts them again and keeps the new count in place of the old, so it is safe to send again; it changes no row. admin.add_columns

Add the new columns of a Saved table's Table reference to the table itself.

It sends DESCRIBE through your query as it is made, and adds each column of the Table reference that the table doesn't have yet. So you add a column by adding its line to the Table reference first, after the table's other columns (the lines of its partition columns may stay last, as they aren't among them): Hive adds new columns after the table's own, and a write fills a table's columns in their order. Then run the Statement it gives, with your execute. The rows already in the table read NULL in a new column until they are written again. For a table with partitioned_by= it writes CASCADE: each Partition already there gets the new columns too. Without it, each of those Partitions would keep its old list of columns, and a Partition written again would lose the new columns' values.

It refuses a table that isn't one of your Saved tables, a Table reference with no new column, a new column that isn't after the table's own, and a new partition column, which a table can't gain: create a new table with it, and copy the rows in.

admin.drop_partitions

Drop some of a Saved table's Partitions: an ALTER TABLE Statement.

Each Partition of the table that where picks goes: every row of it, so no Statement reads it again. It is how a Saved table keeps only so many Partitions, and does the job TRUNCATE can't: Hive refuses TRUNCATE on an external table, a table whose files are in a folder you give, and every table create_table makes is one. The Partitions' files go too when the table has external.table.purge true, which purge=True in its Table reference asks for (help(admin.drop_table)); otherwise they stay in the table's folder.

where picks Partitions by the table's partition columns, as WHERE(...) picks rows: a condition, or a list of conditions that must all hold, each comparing a partition column with a value: equals, not_equals, at_least, at_most, more_than, less_than or between, or all_of(...) of them. Hive takes these in DROP PARTITION, so one Statement drops every Partition they pick. It refuses what DROP PARTITION can't take: any_of(...), a column that isn't one of the table's partition columns, a comparison with another column or a calculation, and any other condition. It refuses a table with no partitioned_by=, and one that isn't one of your Saved tables. The Example database, which is never written to, doesn't run it.

admin.drop_table

Drop one of your Saved tables: the DROP TABLE IF EXISTS Statement.

The warehouse forgets the table, and every Statement that reads it fails until it is created again. Whether its files go too is up to the table itself: they go, every Partition of it, when the table has external.table.purge true, which purge=True in its Table reference asks for. create_table writes it into the table it makes; for a table made before you gave its Table reference purge=True, send admin.set_properties(t) first. Otherwise the warehouse only forgets the table: its files stay in its folder, and a table created there again gets their Partitions back once repair_partitions(t) lists them, or adds to a Partition's old rows with INSERT_INTO; one with no partitioned_by= reads them back at once. The usual reason is a change to a Saved table's columns that admin.add_columns can't make, such as removing or renaming one: drop the table, create it again, and write its rows again, with purge=True set in the table first, or the old rows come back. IF EXISTS means that sending it for a table that isn't there does nothing, rather than failing.

admin.move_to_own_folder

Move a Saved table to a folder of its own: a list of Statements to run.

A Saved table belongs in its own folder, the one its Table reference's location= names: in a folder it shares, two tables read each other's files, and an INSERT OVERWRITE of a Partition in one can delete the other's files for that Partition. This sends DESCRIBE FORMATTED through your query to find where the table is, and returns the Statements that move it, to send in order with run, one at a time:

  • create a copy, named like the table with _moving added, at the folder the Table reference gives the table, its own, stored as the Table reference says;
  • copy the whole table into it with one INSERT OVERWRITE, which, for a table with partitioned_by=, fills each Partition from the rows' own values, as any write does, and needs the Session setting "dynamic_partitions" sent first;
  • compare the rows in the two, Partition by Partition, and stop the move, before anything is dropped, if any Partition differs or is missing from the copy, such as one written since the copy;
  • set external.table.purge to false on the old table, whose folder may hold other tables' files, so that dropping it leaves them. A managed table, one whose files the warehouse keeps where it chooses, as create_table made them before 5.0, skips this step: its files are its own, and go with it;
  • drop the old table;
  • give the copy the table's name: its files stay at its own folder.

So the move takes six Statements, moves[0] to moves[5], or five for a managed table. Each can be sent again until the old table is dropped: if one fails, fix what it says, and send them again from the first. The rows are copied as the old table reads them, so where two tables shared a folder, each read the other's rows too: check them first, or write the table again from its source instead. It refuses a table that isn't one of your Saved tables, one already at its own folder, and one that lacks a column of its Table reference. The Example database is never written to, so it doesn't run the steps.
