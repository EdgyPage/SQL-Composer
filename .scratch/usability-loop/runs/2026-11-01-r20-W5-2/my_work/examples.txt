def runs_and_days_per_job() -> Statement:
    """How many runs each job had, and on how many different days (COUNT DISTINCT).

    count_rows counts every run; count_distinct counts each day once, however many runs
    fell on it.
    """
    return statement(
        SELECT(
            job_runs.job_id,
            AS(count_rows(), "runs"),
            AS(count_distinct(job_runs.dt), "days_with_runs"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
    )

def busy_jobs() -> Statement:
    """Only the jobs with 3 runs or more (HAVING).

    WHERE picks rows before they are counted, so it can't test a count. HAVING tests each
    group after GROUP_BY has counted it.
    """
    return statement(
        SELECT(job_runs.job_id, AS(count_rows(), "runs")),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
        HAVING(at_least(count_rows(), 3)),
    )

two_longest_running_jobs()
def two_longest_running_jobs() -> Statement:
    """The 2 jobs with the most minutes in total (ORDER BY with LIMIT).

    A sort needs a LIMIT: without one, the warehouse puts every row in order before any comes
    back, so the Toolbox refuses a sort that would bring back every row. To sort a whole
    result, sort it in pandas.
    """
    return statement(
        SELECT(job_runs.job_id, AS(sum_of(job_runs.duration_mins), "minutes")),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
        ORDER_BY(descending("minutes")),
        LIMIT(2),
    )

Jobs with no runs, and jobs with at least one: anti-join and semi-join.

Why: a job missing from the runs is often the very problem you're looking for, and a plain JOIN can't show it, since JOIN keeps only the rows that match.

The top of its script, jobs_that_never_ran.py, a Table notebook's logic cell in a Project

    equals, example_database, is_null, statement,
)

jobs_that_never_ran()
def jobs_that_never_ran() -> Statement:
    """Every job with no run on these days (in SQL, an anti-join).

    LEFT_JOIN keeps every job, and a job with no run gets NULL in job_runs' columns, so
    is_null(job_runs.run_id) keeps just those. The days' filter goes in ON=, not in WHERE: in
    WHERE it would throw away the very jobs with no runs, whose dt is NULL too.
    """
    return statement(
        SELECT(jobs.job_id, jobs.job_name),
        FROM(jobs),
        LEFT_JOIN(
            job_runs,
            ON=all_of(equals(job_runs.job_id, jobs.job_id),
                      between(job_runs.dt, FIRST_DAY, LAST_DAY)),
            # A job has many runs. That is fine here: the jobs kept are those with none.
            many_matches=True,
        ),
        WHERE(is_null(job_runs.run_id)),
    )
Hive
SELECT
  jobs.job_id,
  jobs.job_name
FROM ops.jobs AS jobs
LEFT JOIN ops.job_runs AS job_runs
  ON job_runs.job_id = jobs.job_id
  AND job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'
WHERE
  job_runs.run_id IS NULL

jobs_that_ran()
def jobs_that_ran() -> Statement:
    """Every job with at least one run on these days, each job once (in SQL, a semi-join).

    A JOIN straight to job_runs would repeat a job once per run. So first make the list of
    job_ids that ran, each once, with SELECT_DISTINCT, then JOIN to that list.
    """
    ran = derived("ran", statement(
        SELECT_DISTINCT(job_runs.job_id),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
    ))
    return statement(
        SELECT(jobs.job_id, jobs.job_name),
        FROM(jobs),
        JOIN(ran, ON=equals(ran.job_id, jobs.job_id)),
    )
Hive

Label with if_else, fill NULL with fill_null, count by condition, and divide.

Why: codes and NULLs make a result hard to read, and a count of each kind side by side gives one row per job instead of one row per job and status.

The top of its script, labels_and_counts.py, a Table notebook's logic cell in a Project

from sqlglot_composer import (
    AS, FROM, GROUP_BY, SELECT, WHERE, Statement, between, count_rows, equals, example_database,
    fill_null, if_else, is_null, more_than, statement,
)

job_runs = example_database.job_runs

FIRST_DAY = "2026-09-23"

labelled_runs()
def labelled_runs() -> Statement:
    """Each run with a readable status and a length label (in SQL, COALESCE and CASE WHEN).

    A run still going has no status yet (NULL), which fill_null shows as RUNNING. if_else
    gives one value where its condition holds and the other where it doesn't.
    """
    return statement(
        SELECT(
            job_runs.run_id,
            AS(fill_null(job_runs.status, "RUNNING"), "status"),
            AS(if_else(more_than(job_runs.duration_mins, 20), "long", "short"), "length"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
    )
Hive
SELECT
  job_runs.run_id,
  COALESCE(job_runs.status, 'RUNNING') AS status,
  CASE WHEN job_runs.duration_mins > 20 THEN 'long' ELSE 'short' END AS length
FROM ops.job_runs AS job_runs
WHERE
  job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'

def outcomes_per_job() -> Statement:
    """One row per job, with a column for each outcome (a pivot).

    count_rows(where=...) counts only the rows where its condition holds, so each column
    counts one kind of run. A run still going is counted with is_null, since equals can't
    find NULL. A TEST run is in none of the three columns, so they needn't add up to the runs.
    """
    return statement(
        SELECT(
            job_runs.job_id,
            AS(count_rows(where=equals(job_runs.status, "SUCCESS")), "succeeded"),
            AS(count_rows(where=equals(job_runs.status, "FAILED")), "failed"),
            AS(count_rows(where=is_null(job_runs.status)), "still_running"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
    )
Hive
SELECT
  job_runs.job_id,
  COUNT(CASE WHEN job_runs.status = 'SUCCESS' THEN 1 END) AS succeeded,
  COUNT(CASE WHEN job_runs.status = 'FAILED' THEN 1 END) AS failed,
  COUNT(CASE WHEN job_runs.status IS NULL THEN 1 END) AS still_running
FROM ops.job_runs AS job_runs
WHERE
  job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'
GROUP BY
  job_runs.job_id

failed_share_per_job()
def failed_share_per_job() -> Statement:
    """Each job's failed runs as a percent of its runs: one count divided by another.

    Divide after GROUP_BY, as one count by another, not an average of per-row values. Times
    100.0 for a percent; / keeps the decimals even between two counts, in Hive
    (DIV is the one that drops them). A division by 0 gives NULL, which pandas shows as NaN;
    COUNT(*) of a group is never 0, so it can't happen here.
    """
    return statement(
        SELECT(
            job_runs.job_id,
            AS(count_rows(where=equals(job_runs.status, "FAILED")) * 100.0 / count_rows(),
               "failed_percent"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(job_runs.job_id),
    )
Hive
SELECT
  job_runs.job_id,
  (
    COUNT(CASE WHEN job_runs.status = 'FAILED' THEN 1 END) * 100.0
  ) / COUNT(*) AS failed_percent
FROM ops.job_runs AS job_runs
WHERE
  job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'
GROUP BY
  job_runs.job_id

Keep a long list and a lookup as constants, and an if/else chain as one CASE.

Why: a list or a lookup pasted into each Statement, as hand-written Hive often has them, must be found and changed in every one, where a constant at the top of the script, named for what it holds, is changed once.

labelled_runs()
def labelled_runs() -> Statement:
    """Each watched run's status as a label, and its length in words (CASE ... WHEN ...).

    mapped looks each status up in STATUS_LABELS. A run still going has no status (NULL),
    which matches no key, so it gets otherwise=, as any status the dict doesn't hold would; to
    give NULL a label of its own, mapped's docstring fills it in first, with fill_null.
    first_match tries its conditions in order and
    gives the value of the first that holds: a run of 40 minutes is "long", though it is more
    than 10 minutes too.

    Before, as Hive written by hand, the lookup and the chain read as below, with the list
    pasted in where WATCHED_JOBS goes; project.convert(...) writes them as this function does:

        CASE status WHEN 'SUCCESS' THEN 'passed' WHEN 'FAILED' THEN 'failed'
                    WHEN 'TEST' THEN 'test run' ELSE 'still running' END AS outcome,
        CASE WHEN duration_mins > 30 THEN 'long' WHEN duration_mins > 10 THEN 'medium'
             ELSE 'short' END AS length
        ... WHERE job_id IN (1, 2)
    """
    return statement(
        SELECT(
            job_runs.run_id,
            AS(mapped(job_runs.status, STATUS_LABELS, otherwise="still running"), "outcome"),
            AS(first_match([(more_than(job_runs.duration_mins, 30), "long"),
                            (more_than(job_runs.duration_mins, 10), "medium")],
                           otherwise="short"), "length"),
        ),
        FROM(job_runs),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY), is_in(job_runs.job_id, WATCHED_JOBS)),
    )
Hive
SELECT
  job_runs.run_id,
  CASE job_runs.status
    WHEN 'SUCCESS'
    THEN 'passed'
    WHEN 'FAILED'
    THEN 'failed'
    WHEN 'TEST'
    THEN 'test run'
    ELSE 'still running'
  END AS outcome,
  CASE
    WHEN job_runs.duration_mins > 30
    THEN 'long'
    WHEN job_runs.duration_mins > 10
    THEN 'medium'
    ELSE 'short'
  END AS length
FROM ops.job_runs AS job_runs
WHERE
  job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24' AND job_runs.job_id IN (1, 2)

Build a Saved table: create it, write a day, add to a day, backfill, rebuild.

Why: a result that is slow to work out, or read by many notebooks, is cheaper to work out once a day and save than to work out again every time someone reads it.

runs_to_review is partitioned by dt, and each write below fills one day's Partition, named with partition={"dt": day}. Each function below is one step. The Example database can't be written to, so the steps show their Hive but no result. def alerted_runs(day: str = LAST_DAY) -> Statement:
    """Step 3: add that day's runs that succeeded but raised a high alert.

    INSERT_INTO keeps the failed runs step 2 wrote, and adds these beside them. They come
    from a second table, run_alerts, which is why they are a second Statement rather than
    another condition in step 2. Send it once, after step 2. Sent twice, these runs would be in
    the day twice; to redo a day, send step 2 again first, which clears it.
    """
    return statement(
        INSERT_INTO(runs_to_review, partition={"dt": day}),
        # A run can raise several high alerts, so DISTINCT keeps each run once.
        SELECT_DISTINCT(job_runs.run_id, job_runs.job_id),
        FROM(run_alerts),
        JOIN(job_runs, ON=equals(job_runs.run_id, run_alerts.run_id)),
        WHERE(
            equals(run_alerts.dt, day),
            equals(job_runs.dt, day),  # the joined table's partition column needs a filter too
            equals(run_alerts.severity, "high"),
            equals(job_runs.status, "SUCCESS"),
        ),
    )

backfill()
def backfill(days: tuple[str, ...] = (FIRST_DAY, LAST_DAY)) -> list[Statement]:
    """Step 4: write past days, oldest first, each the way steps 2 and 3 write one.

    To backfill is to write days that have already happened: a new table's past days, every
    day again after rebuild(), or a day again after fixing a mistake in a step. Each day
    gets step 2 then step 3, in that order, so it starts clean and keeps both sources. A
    table filled by one Statement alone can instead be written in one go: INSERT_OVERWRITE
    without partition=, its SELECT listing dt last, fills each day's Partition from the rows.

        for s in backfill():
            run_execute(s, execute)
    """
    return [s for day in days for s in (failed_runs(day), alerted_runs(day))]

rebuild()
def rebuild() -> list[Statement]:
    """Step 5, rarely: after changing the table's columns, drop it and create it again.

    create_table refuses while the old table exists, so the old one has to go first.
    admin.drop_table deletes every day of the table, since its Table reference had purge=True
    when create_table made it, so write them all again with backfill(). A column only added
    after the others can instead be added in place with admin.add_columns, without dropping
    anything; on a partitioned table it writes CASCADE, so the Partitions already there get
    the column too.
    IF EXISTS means the drop does nothing, rather than failing, if the table isn't there.
    """
    return [admin.drop_table(runs_to_review), create_table(runs_to_review)]

def minutes_per_team() -> Statement:
    """Step 2: add each job's team from jobs, then total the minutes per team.

    Summing step 1's minutes again is safe, since a sum of sums is still the right sum. The

def careless(keeps_only_matches: bool = False) -> Statement:
    """What most people write first: the days go in WHERE. The Guard refuses it.

    careless(keeps_only_matches=True) pastes the Guard's opt-out and shows what goes wrong:
    three jobs come back, and cache_warm, with no runs, is missing.
    """
    return statement(
        SELECT(jobs.job_name, AS(count_rows(where=is_not_null(job_runs.run_id)), "runs")),
        FROM(jobs),
        # Each job matches several runs, and counting them is the point: many_matches=True.
        LEFT_JOIN(job_runs, ON=equals(job_runs.job_id, jobs.job_id),
                  many_matches=True, keeps_only_matches=keeps_only_matches),
        WHERE(between(job_runs.dt, FIRST_DAY, LAST_DAY)),
        GROUP_BY(jobs.job_name),
    )

A Guard refuses it

GuardRefused:
  What happened:  WHERE has job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24', a condition on job_runs, which LEFT_JOIN brought in.
  Why it matters: LEFT_JOIN keeps rows with no match in job_runs, but those rows have NULL there, so this WHERE throws them away again. The result is an ordinary JOIN.
  Usual fix:      Move the condition into LEFT_JOIN(job_runs, ON=...), next to the join condition, so it only decides which rows match.
  Opt-out:        LEFT_JOIN(job_runs, ON=..., keeps_only_matches=True)

Table

A Table reference: a table's columns, its partition columns and its key.

run_alerts = Table("ops.run_alerts", partitioned_by=["dt"], key=["alert_id"],
    columns={"alert_id": "bigint", "severity": "string", "dt": "string"})

Python shows

Table('ops.run_alerts', columns: alert_id, severity, dt)

Its columns become attributes, so job_runs.status is checked when your script runs: a typo stops with the list of real columns. Write one per table, generated once by write_table_reference(...) and then yours to edit.

columns maps each column to its Hive type as DESCRIBE, the command that lists a table's columns, prints it ("bigint", "decimal(10,2)"), or None when you don't know it. A typed column gets its values checked.
partitioned_by lists the columns the table is partitioned by, in the order Hive partitions it, such as ["dt"] or ["region", "load_id"], each a column in columns too; leave it out for a table that isn't partitioned. Each value of them is a folder of the table's files, a Partition, and a Statement reading the table must filter at least one of them (help(FROM)). A write fills the Partitions its rows hold, or the ones it names (help(INSERT_OVERWRITE)).
key lists the columns that pick out one row, the table's Key, for a table that has one; leave it out for one that hasn't. JOIN warns when a join doesn't use all of them; a write into a Saved table with a Key is refused unless the Toolbox can see each row it writes has a Key of its own (help(INSERT_OVERWRITE) says how); and create_table writes it into the table (help(create_table)). Hive never checks it.
does_not_add_up lists columns that are averages, ratios or distinct counts, which sum_of(...) and average_of(...) refuse to add up.
descriptions says in words what a column holds, such as {"status": "SUCCESS or FAILED"}, for the columns you want to describe. write_table_reference fills it from the comments DESCRIBE prints; it never changes the Hive.

A Saved table, one your Statements write into, also says how it is stored, which create_table writes and check_table_reference compares with the table. Of these, a Saved table needs only location; a table someone else owns has none of them:

location is the table's own folder, exactly as create_table writes it, such as "/data/your_team/mart.db/daily_runs": the Toolbox adds nothing to it. Every Saved table is an external table: its files are yours, in that folder, and the warehouse only keeps a record of where. Two tables in one folder would read each other's files, so a Saved table whose folder is another's, or inside another's, is refused. A location such as "${hivevar:root}/mart.db/daily_runs" holds a Hive variable: send its SET first, with run_execute(hive_settings("root", sets={"root": {"hivevar:root": "/data/your_team"}}), execute), and check_table_reference then notes the table's folder rather than comparing it.
purge=True makes dropping the table delete its files too. purge=False keeps them in its folder, where a table created again reads them back: at once, or, for a partitioned table, its Partitions once repair_partitions(t) lists them; left out, the warehouse's own choice, which keeps them, is neither written nor checked.
properties holds any other table properties, the notes on a table the warehouse keeps by name, such as {"made_by": "daily_pipeline"}.
stored_as is the file format, "ORC" (the usual one) or "PARQUET", and compression how its files are packed: "SNAPPY", or ORC's "ZLIB" or Parquet's "GZIP", among others.
bucketed_by with buckets spreads each Partition's rows into that many files by the values of those columns, and sorted_by sorts the rows in each, so a join on them can pair file with file. Leave them out unless the table is big and joined on those columns: changing them means writing the table again.

A column whose name is a Python word, such as from, is reached with getattr(t, "from"). You never type backticks: the Hive puts a name in them where it needs them, such as from, any or two words, and leaves a plain name, such as dt, as it is.

A table's text, in an f-string or from str(...), is the table as FROM writes it, ops.jobs AS jobs, for after FROM or JOIN in Hive you write by hand; DESCRIBE and SHOW PARTITIONS take the name alone, ops.jobs, which you write yourself. A column's text is the column as SELECT writes it, jobs.job_id, not ops.jobs.job_id, even where you reach it as ops.jobs.job_id: Hive names a column by its table's name after AS, and refuses a three-part name in a SELECT. help(unchecked_hive) shows them in use. A format spec works on them as on a str:

f"SELECT {jobs.job_id} FROM {jobs}"

A Saved table is one your Statements write into. Send it once with run_execute(create_table(t), execute). You rarely need the hint, though: with "map_joins" sent, Hive makes map joins by itself, from the tables' sizes and statistics. Ask for it only for a table you know is small: copied to every worker, a big one can run them out of memory.

FULL_JOIN

Pair two tables' matching rows, and keep each table's rows with no match.

recorded = derived("recorded", statement(
    SELECT(job_runs.run_id), FROM(job_runs), WHERE(last_n_days(job_runs.dt, 2))))

scheduler_runs = example_database.scheduler_runs

started = derived("started", statement(
    SELECT(scheduler_runs.run_id), FROM(scheduler_runs),
    WHERE(last_n_days(scheduler_runs.dt, 2))))

print(to_hive(statement(
    SELECT(AS(recorded.run_id, "in_job_runs"),
           AS(started.run_id, "in_scheduler_runs")),
    FROM(recorded),
    FULL_JOIN(started, ON=equals(started.run_id, recorded.run_id)),
    WHERE(any_of(is_null(recorded.run_id), is_null(started.run_id))),
)))

Hive

WITH recorded AS (
  SELECT
    job_runs.run_id
  FROM ops.job_runs AS job_runs
  WHERE
    job_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'
), started AS (
  SELECT
    scheduler_runs.run_id
  FROM ops.scheduler_runs AS scheduler_runs
  WHERE
    scheduler_runs.dt BETWEEN '2026-09-23' AND '2026-09-24'
)
SELECT
  recorded.run_id AS in_job_runs,
  started.run_id AS in_scheduler_runs
FROM recorded
FULL JOIN started
  ON started.run_id = recorded.run_id
WHERE
  recorded.run_id IS NULL OR started.run_id IS NULL

Use it to compare two records of the same things, such as the runs job_runs holds and the runs the scheduler started: each run comes out once, matched or not, with NULL in the columns of the side that lacks it, and fill_null(...) takes a value from whichever side has one. A WHERE on either side's columns would throw away the rows the other side lacks, so it is refused, unless it keeps them: is_null(...) of a column, or any_of(...) with such an is_null among its conditions. keeps_only_matches=True lets such a WHERE through, for when losing those rows is what you mean. For the same reason, read a partitioned table, on either side, through a derived(...) table that filters its partition columns in its own WHERE, and FULL_JOIN that: ON= can't filter it, since FULL_JOIN keeps the rows ON= doesn't match anyway. Either side's columns may be NULL, so its rows have no Key the Toolbox can see: a write needs a GROUP_BY to give them one, such as GROUP_BY(fill_null(recorded.run_id, started.run_id)) for the example above. A repeated-rows Warning, and its many_matches=True, work as JOIN's do. The example above gives the runs only one side holds.

In a date_format pattern, write yyyy for the year, not YYYY: YYYY is the year a week belongs to, which differs from the year in the last days of December and the first of January.

It checks the call before writing it:

it refuses a call with the wrong number of arguments, for the functions it knows, such as upper, substr or date_add;
a function that turns many rows into one, such as collect_set or percentile, is treated as count_rows() is: it needs a name with AS, and GROUP_BY for the other columns;
it refuses a function that works only over a window of rows, written OVER (...) after it, such as lag or rank, since it can't write OVER. For lag, lead, rank, dense_rank, ntile and row_number, the refusal names the Toolbox's own function, such as lag(...), which takes a window(...); for the others, such as percent_rank, it says to work it out in pandas.

window

The rows a calculation looks across for each row, passed to it as over=.

Hive

PARTITION_BY= makes the groups: each row looks only at the rows sharing its values. The word is SQL's, and has nothing to do with the Partitions a table is stored in. Left out, every row is in one group.
ORDER_BY= puts each group's rows in order, with descending(...) for largest first.
rows=(start, end) keeps only the rows near each one, counted from the row itself: negative before it, 0 the row itself, positive after it. None means no limit: every row before it as the start, every row after it as the end. rows=(-6, 0) is the row and the 6 before it; rows=(None, 0) every row up to it. It needs ORDER_BY. Left out, each row looks across its whole group.

PARTITION_BY and ORDER_BY take columns or a list of them, never a name given in SELECT. Each row keeps its own row, so a Window changes no row count, and needs no GROUP_BY. Hive can't test a Window's result in WHERE or HAVING of the Statement that works it out: work it out in a derived(...) table, and test it in the Statement that reads it. A Window over the totals of a GROUP_BY, such as each job's share of the day, goes the same way: total in a derived(...) table, and the Window in the Statement that reads it.

Hive keeps a Session setting for the rest of the session, so send them first, then your Statements: run_execute(hive_settings(...), execute) sends each SET line through your execute, as a call of its own, and gives back None. They hold only while your query and execute share one session across their calls: if each call opens a new session, each SET is gone before your Statement runs. show_hive prints them, and export_lineage, passed them with the Statements, writes them at the foot of both its files.

Name one or more sets. The Toolbox ships four:

"dynamic_partitions": hive.exec.dynamic.partition=true and hive.exec.dynamic.partition.mode=nonstrict. A write that fills a partition column from its rows' own values, as INSERT_OVERWRITE(t) does for each one partition= doesn't name, runs only with these on; show_hive prints them ahead of such a write.
"map_joins": hive.auto.convert.join=true, hive.auto.convert.join.noconditionaltask=true and hive.ignore.mapjoin.hint=false. "statistics_on_write": hive.stats.autogather=true and hive.stats.column.autogather=true. Each write then gathers the statistics of the Partitions it writes, which Hive picks its join plans from, so they are never out of date. compute_statistics(t, partition=...) counts a Partition written without them.
"merge_small_files": hive.merge.mapfiles=true, hive.merge.mapredfiles=true and hive.merge.tezfiles=true. A write then merges its small files into fewer, larger ones, since many small files slow down every later read.

A Decision made already, as yours is once you change its line, is kept, and an answer that differs from it is named; so is an answer for a table with no Table reference, or one that answers no Decision. A converted table's conversion/checklist.md, which lists its open Decisions, is written again. Its Workflow's next step names each Decision still open.
