# sqlglot is deliberately not semver: minor releases carry backwards-incompatible changes,
# and this library leans on behaviour its documentation does not contract - Hive
# string-literal escaping, qualify() raising OptimizeError on an unknown column, and
# ErrorLevel.RAISE refusing a construct Hive cannot express. An upgrade is a deliberate
# change with the escaping matrix in tests/test_escaping.py as its gate, never a float.
# See docs/adr/0001-sqlglot-over-sqlalchemy.md.
sqlglot==30.18.0

# The Lineage graph, and the git-tracked static Lineage artifact derived from it. Pinned
# because that artifact is committed and diffed in review: a traversal order that shifted
# between releases would show every line of every artifact as changed.
networkx==3.1

# ------------------------------------------------------------------------------------
# TEST ONLY. NOTHING UNDER sqlcomposer/ IMPORTS ANY OF THESE, and the suite has a test
# that says so. The core is compile-only: it returns SQL strings and Lineage, and never
# connects to anything. These are declared as the `test` extra in pyproject.toml, so
# `pip install .` does not pull them in.
# ------------------------------------------------------------------------------------

# Runs the emitted Hive SQL so the value tests can check numbers rather than text.

duckdb==1.5.5

# The frame type a caller's Runner hands back. runner.py imports it under TYPE_CHECKING

numpy==1.25.2

pytest==9.0.2
